#### 2.2.1 主要数据表结构

```sql
-- 用户表
CREATE TABLE users (
    id BIGINT PRIMARY KEY AUTO_INCREMENT,
    username VARCHAR(50) UNIQUE NOT NULL,
    email VARCHAR(100) UNIQUE,
    phone VARCHAR(20) UNIQUE,
    password_hash VARCHAR(255) NOT NULL,
    avatar_url VARCHAR(255),

);

-- 账本表
CREATE TABLE account_books (
    id BIGINT PRIMARY KEY AUTO_INCREMENT,
    user_id BIGINT NOT NULL,
    name VARCHAR(100) NOT NULL,
    description TEXT,
    currency VARCHAR(10) DEFAULT 'CNY',
    month_start_day TINYINT DEFAULT 1 COMMENT '月度开始日期',
    is_default BOOLEAN DEFAULT FALSE,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    INDEX idx_user_id (user_id)
);

-- 分类表
CREATE TABLE categories (
    id BIGINT PRIMARY KEY AUTO_INCREMENT,
    account_book_id BIGINT NOT NULL,
    parent_id BIGINT DEFAULT NULL,
    name VARCHAR(50) NOT NULL,
    type ENUM('INCOME', 'EXPENSE') NOT NULL,
    icon VARCHAR(50),
    color VARCHAR(20),
    sort_order INT DEFAULT 0,
    is_system BOOLEAN DEFAULT FALSE,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    INDEX idx_account_book_id (account_book_id),
    INDEX idx_parent_id (parent_id)
);

-- 账户表
CREATE TABLE accounts (
    id BIGINT PRIMARY KEY AUTO_INCREMENT,
    account_book_id BIGINT NOT NULL,
    name VARCHAR(100) NOT NULL,
    type ENUM('CASH', 'BANK_CARD', 'CREDIT_CARD', 'ALIPAY', 'WECHAT', 'OTHER') NOT NULL,
    balance DECIMAL(15,2) DEFAULT 0.00,
    initial_balance DECIMAL(15,2) DEFAULT 0.00,
    is_included_in_total BOOLEAN DEFAULT TRUE,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    INDEX idx_account_book_id (account_book_id)
);

-- 交易记录表
CREATE TABLE transactions (
    id BIGINT PRIMARY KEY AUTO_INCREMENT,
    account_book_id BIGINT NOT NULL,
    account_id BIGINT NOT NULL,
    category_id BIGINT NOT NULL,
    type ENUM('INCOME', 'EXPENSE', 'TRANSFER') NOT NULL,
    amount DECIMAL(15,2) NOT NULL,
    description TEXT,
    transaction_date DATE NOT NULL,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    INDEX idx_account_book_id (account_book_id),
    INDEX idx_account_id (account_id),
    INDEX idx_category_id (category_id),
    INDEX idx_transaction_date (transaction_date)
);

-- 标签表
CREATE TABLE tags (
    id BIGINT PRIMARY KEY AUTO_INCREMENT,
    account_book_id BIGINT NOT NULL,
    name VARCHAR(50) NOT NULL,
    color VARCHAR(20),
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    INDEX idx_account_book_id (account_book_id)
);

-- 交易标签关联表
CREATE TABLE transaction_tags (
    transaction_id BIGINT NOT NULL,
    tag_id BIGINT NOT NULL,
    PRIMARY KEY (transaction_id, tag_id)
);

-- 预算表
CREATE TABLE budgets (
    id BIGINT PRIMARY KEY AUTO_INCREMENT,
    account_book_id BIGINT NOT NULL,
    category_id BIGINT,
    name VARCHAR(100) NOT NULL,
    amount DECIMAL(15,2) NOT NULL,
    period_type ENUM('MONTHLY', 'YEARLY') NOT NULL,
    start_date DATE NOT NULL,
    end_date DATE,
    alert_threshold DECIMAL(5,2) DEFAULT 0.8,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    INDEX idx_account_book_id (account_book_id)
);
```

```java
@Service
public class VoiceRecognitionService {
    
    @Autowired
    private BaiduAiClient baiduAiClient;
    
    @Autowired
    private NlpService nlpService;
    
    public TransactionDTO recognizeVoice(MultipartFile audioFile) {
        // 1. 语音转文字
        String text = baiduAiClient.speechToText(audioFile);
        
        // 2. 自然语言处理
        ParseResult parseResult = nlpService.parseTransaction(text);
        
        // 3. 构建交易记录
        return TransactionDTO.builder()
            .amount(parseResult.getAmount())
            .categoryId(parseResult.getCategoryId())
            .description(parseResult.getDescription())
            .transactionDate(parseResult.getDate())
            .build();
    }
}
```

#### 3.2.1 统计服务实现
```java
@Service
public class StatisticsService {
    
    public MonthlyStatistics getMonthlyStatistics(Long accountBookId, LocalDate month) {
        // 根据账本的月度开始日计算统计周期
        AccountBook accountBook = accountBookService.getById(accountBookId);
        LocalDate startDate = month.withDayOfMonth(accountBook.getMonthStartDay());
        LocalDate endDate = startDate.plusMonths(1).minusDays(1);
        
        // 查询期间内的交易记录
        List<Transaction> transactions = transactionService.getByDateRange(
            accountBookId, startDate, endDate);
        
        return calculateStatistics(transactions);
    }
}
```

### 3.3 数据同步功能

#### 3.3.1 同步策略
```java
@Component
public class DataSyncService {
    
    @Autowired
    private RedisTemplate<String, Object> redisTemplate;
    
    // 增量同步
    public void syncData(Long userId, SyncRequest request) {
        String lastSyncTime = request.getLastSyncTime();
        
        // 获取增量数据
        List<Transaction> changedTransactions = transactionService
            .getChangedSince(userId, lastSyncTime);
        
        // 冲突检测和解决
        List<Transaction> resolvedTransactions = conflictResolver
            .resolve(changedTransactions, request.getClientData());
        
        // 返回同步结果
        return SyncResponse.builder()
            .data(resolvedTransactions)
            .serverTime(System.currentTimeMillis())
            .build();
    }
}
```

### 5.1 身份认证与授权
```java
@Configuration
@EnableWebSecurity
public class SecurityConfig {
    
    @Bean
    public SecurityFilterChain filterChain(HttpSecurity http) throws Exception {
        return http
            .authorizeHttpRequests(auth -> auth
                .requestMatchers("/auth/**").permitAll()
                .anyRequest().authenticated())
            .oauth2ResourceServer(oauth2 -> oauth2
                .jwt(jwt -> jwt.jwtDecoder(jwtDecoder())))
            .build();
    }
}
```
